ALGORITHME Rectangle
VARIABLES                   
        lg, lr, s, p : RÉEL

    LIRE lr
    s ← lg * lr
    p ← 2 * (lg + lr)
    AFFICHER "La surface est de : ", s
    AFFICHER "Le périmètre est de : ", p
